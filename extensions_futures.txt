from collections import defaultdict
import datetime
import subprocess


def audit_git_repo():
    # 1. Analyse des commits, fichiers touchés et dates par auteur
    cmd = [
        "git",
        "log",
        "--pretty=format:%an|%ad|%s",
        "--date=short",
        "--numstat",
    ]
    res = subprocess.run(cmd, capture_output=True, text=True, check=True)

    author_stats = defaultdict(
        lambda: {
            "commits": 0,
            "dates": set(),
            "code_lines": 0,
            "test_lines": 0,
            "test_files": 0,
        }
    )

    current_author = None
    for line in res.stdout.splitlines():
        if "|" in line and not line.startswith("\t"):
            parts = line.split("|")
            current_author = parts[0].strip()
            date = parts[1].strip()
            author_stats[current_author]["commits"] += 1
            author_stats[current_author]["dates"].add(date)
        elif line.strip() and current_author:
            chunks = line.split()
            if len(chunks) == 3 and chunks[0].isdigit() and chunks[1].isdigit():
                added, deleted, filepath = (
                    int(chunks[0]),
                    int(chunks[1]),
                    chunks[2],
                )
                # On ignore les fichiers de cache, données et lockfile
                if any(
                    x in filepath for x in ["data/", "cache/", "poetry.lock"]
                ):
                    continue
                if "tests/" in filepath:
                    author_stats[current_author]["test_lines"] += added
                    author_stats[current_author]["test_files"] += 1
                else:
                    author_stats[current_author]["code_lines"] += added

    # Affichage du diagnostic
    print(
        f"{'Auteur':<20} | {'Commits':<8} | {'Jours actifs':<12} | {'Lignes Code':<12} | {'Lignes Tests'}"
    )
    print("-" * 75)
    for author, s in author_stats.items():
        print(
            f"{author:<20} | {s['commits']:<8} | {len(s['dates']):<12} | {s['code_lines']:<12} | {s['test_lines']} ({s['test_files']} fichiers)"
        )


if __name__ == "__main__":
    audit_git_repo()
